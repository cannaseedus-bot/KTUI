cmake_minimum_required(VERSION 3.16)
project(kuhul_micronaut_grammar LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(kuhul_grammar STATIC grammar.cpp)
target_include_directories(kuhul_grammar PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(micronaut-parse grammar_main.cpp)
target_link_libraries(micronaut-parse PRIVATE kuhul_grammar)
add_executable(grammar-test test_grammar.cpp)
target_link_libraries(grammar-test PRIVATE kuhul_grammar)
enable_testing()
add_test(NAME grammar-test COMMAND grammar-test)
